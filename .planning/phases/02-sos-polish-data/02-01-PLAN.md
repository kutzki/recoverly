async function logCTAAction(actionId: string) {
  if (!incidentIdRef.current) return;
  try {
    const { data: existing } = await supabase
      .from('crisis_incidents')
      .select('actions_completed')
      .eq('id', incidentIdRef.current)
      .single();

    await supabase

}
